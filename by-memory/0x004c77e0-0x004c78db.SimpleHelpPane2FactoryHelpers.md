static int CreateSimpleHelpPane2FromAnchorRect(
    const char* text,
    Pane* owner,
    const RectBounds* anchor,
    int timeout,
    int styleA,
    int styleB,
    int styleC)
{
    const int x = (anchor->left + anchor->right) / 2;
    const int y = anchor->top - 6;
    return CreateSimpleHelpPane2(text, owner, x, y, anchor, timeout, styleA, styleB, styleC);
}

bool SimpleHelpPane2::AnchorNeedsRefresh()
{
    Point cursor;
    g_pEventMan->GetCursorPosition(&cursor);

    Point screenOffset;
    GetScreenOffset(&screenOffset);

    RectBounds current = m_anchorRect;
    current.Offset(screenOffset.x, screenOffset.y);

    if (m_anchorRect.IsEmptyOrInvalid())
        return true;

    return PointInRect(cursor.y, cursor.x, &current);
}
