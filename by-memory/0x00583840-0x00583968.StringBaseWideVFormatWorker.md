        RefCountedStringBufferHeader *header =
            reinterpret_cast<RefCountedStringBufferHeader *>(m_data) - 1;

            AllocateWideBuffer(requested);

                reinterpret_cast<RefCountedStringBufferHeader *>(m_data) - 1;

        m_data[capacity] = L'\0';
        header->length = capacity;

                                      capacity,
                                      capacity,

                                      args);
        if (result != -1) {
            break;
        }

        requested = capacity * 2;
    }

    if (m_data != g_emptyWideString) {
        RefCountedStringBufferHeader *header =
            reinterpret_cast<RefCountedStringBufferHeader *>(m_data) - 1;
