    popupRect.right = (popupTextWidth > 0) ? anchorX + popupTextWidth + 4 : anchorX;

    const int screenWidth = g_pScreenPane->GetScreenWidth();
    const int screenHeight = g_pScreenPane->GetScreenHeight();

    if (popupRect.right >= screenWidth) {
