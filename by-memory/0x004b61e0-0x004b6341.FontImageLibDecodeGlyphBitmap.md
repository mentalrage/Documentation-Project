                                    bool clearBeforeDecode,
                                    const unsigned char *paletteMap,

{

    tileContext->rowStridePixels = tileContext->bounds.right - tileContext->bounds.left;

    const int requiredBytes = RectArea(&tileContext->bounds);

        if (m_scratchBuffer != NULL) {
            m_scratchBuffer = static_cast<unsigned char *>(
                GetMemoryMan()->ReallocateBufferMemory(m_scratchBuffer, requiredBytes));
        } else {
            m_scratchBuffer = static_cast<unsigned char *>(
                GetMemoryMan()->AllocateBufferMemory(requiredBytes));
        }
        m_scratchBufferBytes = requiredBytes;
    }

    if (clearBeforeDecode) {
        memset(m_scratchBuffer, 0, m_scratchBufferBytes);
    }

    tileContext->pixelData = m_scratchBuffer;
    tileContext->encodedMaskByteCount = 0;
    tileContext->encodedMaskBytes = 0;

    const unsigned char *encoded = glyph->encodedRows;
    unsigned char *row = m_scratchBuffer;
    const int width = tileContext->bounds.right - tileContext->bounds.left;
    const int height = tileContext->bounds.bottom - tileContext->bounds.top;

    for (int y = 0; y < height; ++y) {
        int x = 0;
        for (;;) {
            const unsigned char token = *encoded++;
            if (token == 0) {
                break;
            }

            const int runStart = x;
            const int runLength = token & 0x1f;
            const int paletteIndex = token >> 5;
            x += runLength;

            if ((!clearBeforeDecode && paletteIndex != 5) ||
                (clearBeforeDecode && paletteIndex != 0)) {
                if (x > 0 && runStart < width) {
                    const int left = runStart > 0 ? runStart : 0;
                    const int right = x < width ? x : width;
                    if (right > left) {
                        memset(row + left, paletteMap[paletteIndex], right - left);
                    }
                }
            }
        }

        row += tileContext->rowStridePixels;
    }

    return height;

## Behavior

This method expands one [UID:0001UK][FontGlyphRecord](by-type/by-struct/FontGlyphRecord.md) into the shared `FontImageLib` scratch buffer and exposes that buffer through an [UID:00004I][EPFTileContext](by-class/EPFTileContext.md)-shaped output context.

Observed flow:

- builds a destination rectangle at `tileContext + 0x10` from the glyph record fields:
  - `record + 0x04`: left;
  - `record + 0x02`: top;
  - `record + 0x08`: right;
  - `record + 0x06`: bottom;
- stores row stride `right - left` at `tileContext + 0x0c`;
- computes required scratch bytes with [UID:00015S][0x004b7c30-0x004b7f87.RectGeometryHelpers](by-memory/0x004b7c30-0x004b7f87.RectGeometryHelpers.md) subrange `0x004b7ca0` / `RectArea`;
- grows the shared scratch buffer at `this + 0x80014` through the MemoryMan allocation helpers when the required byte count exceeds `this + 0x80018`;
- optionally clears the entire current scratch capacity before decoding;
- stores the scratch pointer at `tileContext + 0x04`;
- clears `tileContext + 0x20` and `tileContext + 0x24`, matching the encoded-mask size/data fields in [UID:00004I][EPFTileContext](by-class/EPFTileContext.md);
- starts encoded row decoding at `record + 0x0e`;
- processes one zero-terminated token stream per output row;
- for each nonzero token:
  - low five bits are run length;
  - high three bits are palette index;
  - the palette index selects a byte from the caller-provided palette/remap table;
  - if the buffer was not pre-cleared, palette index `5` is the transparent/skip token and other indexes write;
  - if the buffer was pre-cleared, palette index `0` is skipped and nonzero indexes write;
- advances to the next encoded row after each `0` terminator.
- returns the decoded rectangle height (`bottom - top`).

Writes are clipped horizontally to the row width when the token stream starts before or extends past the valid row span. The code assumes the compressed row stream is otherwise well-formed enough that row runs do not require defensive bounds checks beyond the observed start/end clipping.
