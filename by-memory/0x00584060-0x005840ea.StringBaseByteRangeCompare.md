                     const char *lhsEnd,
                     const char *rhsBegin,

{

    const char *rhs = rhsBegin;

    while (lhs != lhsEnd && rhs != rhsEnd) {

        const unsigned char rhsChar = static_cast<unsigned char>(*rhs);
