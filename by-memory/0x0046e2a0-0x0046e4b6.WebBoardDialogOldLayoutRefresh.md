int __thiscall WebBoardDialogOld::RefreshOldLayout(int mode)
{
    Rect closeRect;
    Rect browserRect;
    Rect dialogRect;

    this->ResetLayoutVirtual38();
    this->ResetLayoutVirtual40();
    this->SetActiveControl(1);
    this->SetActiveControl(0);

    if (this->closeButton)
        this->closeButton->Destroy(1);
    if (this->browserPane)
        this->browserPane->Destroy(1);

    else if (mode == 6)
        closeRect.Set(368, 568, 431, 592);

    this->closeButton = ConstructImageButtonControl(0x114, 14, closeRect);
    this->AttachChild(this->closeButton);

    if (mode == 0)
        browserRect.Set(11, 42, 689, 460);
    else if (mode == 4)
        browserRect.Set(11, 42, 407, 526);
    else if (mode == 6)
        browserRect.Set(11, 42, 789, 560);

    this->browserPane = ConstructBrowserControlPane(0x338, browserRect, this);
    this->AttachChild(this->browserPane);

    if (mode == 0)
        dialogRect.Set(162, 134, 862, 634);
    else if (mode == 4)
        dialogRect.Set(303, 101, 721, 667);
    else if (mode == 6)
        dialogRect.Set(112, 84, 912, 684);

    this->SetBounds(dialogRect, 0, 0, dword_69B374);
    this->SetScrollOrOrigin(0, 0);
    this->RefreshAfterLayout();
    this->SetCurrentControl(0);
