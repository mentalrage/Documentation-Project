                                             DDLOCK_WAIT,
                                             NULL) == DD_OK) {
                    WORD *sourcePixels =
                        static_cast<WORD *>(sourceDescription.lpSurface);
                    DWORD *destinationPixels =
                        static_cast<DWORD *>(destinationDescription.lpSurface);
                    const int sourcePitch = sourceDescription.lPitch / sizeof(WORD);
                    const int destinationPitch =
                        destinationDescription.lPitch / sizeof(DWORD);

                    for (unsigned short y = 0;
                         y < sourceDescription.dwHeight;

                        for (unsigned short x = 0;
                             x < sourceDescription.dwWidth;
                             ++x) {

                                ((pixel & 0x07e0) << 5) |
                                ((pixel & 0x001f) << 3);
                        }
                    }

                    m_offscreenSurface->Unlock(destinationDescription.lpSurface);
                }

                sourceSurface->Unlock(sourceDescription.lpSurface);

            const int windowLeft = g_pApplication->m_windowLeft;
            const int windowHeight = g_pApplication->m_windowHeight;
            const int windowWidth = g_pApplication->m_windowWidth;

            if (g_displayStartupDrawType == 1) {
                RECT windowRect = {
                    windowLeft,
                    windowTop,
                    windowLeft + windowWidth,
                    windowTop + windowHeight

                HRESULT result = m_primarySurface->Blt(&windowRect,

                                                       &fullSurfaceRect,
                                                       DDBLT_WAIT,

                       windowTop,

## Behavior

The method presents a supplied ScreenPane render surface through four exact runtime paths:

1. When `g_isWindowedDisplayMode == 0`, it calls `m_primarySurface->BltFast(sourceRect->left, sourceRect->top, sourceSurface, sourceRect, DDBLTFAST_WAIT)`. Only `DDERR_SURFACELOST` is handled: it calls `Restore()` without checking that result and retries `BltFast` exactly once.
2. In windowed mode with `g_surfaceUses32BitPresentation == 0`, it copies the input RECT and calls `m_primarySurface->Blt(&destinationRect, sourceSurface, sourceRect, DDBLT_WAIT, NULL)`. Its result is ignored; there is no lost-surface retry on this branch.
3. In windowed 32-bit mode it locks `sourceSurface` and `m_offscreenSurface` with `DDLOCK_WAIT`, converts RGB565 into zero-alpha `0x00RRGGBB`, then uses DirectDraw `Blt` when `g_displayStartupDrawType == 1`. The destination rectangle is built from `Application+0xa58/+0xa5c/+0xa60/+0xa64` as top/left/height/width, and the source rectangle is `{0,0,g_screenWidth,g_screenHeight}`. `DDERR_SURFACELOST` produces one unchecked `Restore()` and one retry.
4. Other windowed 32-bit draw types obtain source and primary-surface DCs without checking `GetDC`. Equal configured and screen dimensions use `BitBlt(..., SRCCOPY)`; otherwise `SetStretchBltMode` selects `COLORONCOLOR` for DrawType 2 and `HALFTONE` for DrawType 3 or any fallback value before `StretchBlt(..., SRCCOPY)`. Source DC is released before destination DC.

The RGB565 conversion is exact: `((pixel & 0xf800) << 8) | ((pixel & 0x07e0) << 5) | ((pixel & 0x001f) << 3)`. It does not replicate low channel bits and does not set alpha. Source and destination pitches are divided by two and four bytes respectively, and the nested counters retain the observed unsigned 16-bit behavior.

For [UID:00039Z][0x00559170-0x005591fc.SurfaceRestoreLostSurfaces](by-memory/0x00559170-0x005591fc.SurfaceRestoreLostSurfaces.md), this helper is support evidence for naming the ScreenPane receiver field at `+0x100` as `m_primarySurface` / `m_displaySurface`: the blit path uses that field as the display/presentation DirectDraw destination and restores it when blit/lost-surface handling observes `DDERR_SURFACELOST` (`0x887601C2`). ## Error And Cleanup Semantics

- Each `Lock` must return `DD_OK` before its dependent body runs. Destination Lock failure still reaches source Unlock.
- `m_offscreenSurface->Unlock(destinationDescription.lpSurface)` occurs only after successful destination Lock; `sourceSurface->Unlock(sourceDescription.lpSurface)` follows it when source Lock succeeded.
- Both `GetDC` return values are ignored. The source DC is released before the primary-surface DC, and no failure compensation is invented.
- Only `DDERR_SURFACELOST` receives special handling. `Restore` results are ignored, retries occur once rather than in a loop, and retry results are not propagated.
- No caller-visible status exists; preserving these weak checks and cleanup asymmetries is required runtime behavior.
