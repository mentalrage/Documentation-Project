void LivingObjectPane::SetMapPosition(int x, int y)
{
    if (m_mapX == x && m_mapY == y)
        return;
    g_activeMapPane->MoveObject(this, m_mapY, m_mapX, y, x);
    Invalidate(false);
    m_mapX = x;
    m_mapY = y;
    Invalidate(false);
    CancelTimer(kLivingTimerForwardStep);
    RefreshAllLinkedObjects();
    m_isMoving = false;
}
