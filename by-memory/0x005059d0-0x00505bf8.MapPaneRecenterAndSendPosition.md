                                      int targetY,
                                      int viewportX,
                                      int viewportY,
                                      bool useActiveObjectViewport)
{
    MapPoint oldViewport;
    oldViewport.column = viewportX;
    oldViewport.row = viewportY;

    if (useActiveObjectViewport && m_localPlayerObjectPane != NULL)
        m_localPlayerObjectPane->GetViewportPosition(&oldViewport);

    if (targetX < 0 || targetY < 0 ||
        targetX >= m_mapWidth || targetY >= m_mapHeight ||
        oldViewport.column < 0 || oldViewport.row < 0 ||
        oldViewport.column >= m_visibleTileColumns ||
        oldViewport.row >= m_visibleTileRows)
    {
        return;
    }

    if (g_useEpfAssets)
    {
        m_viewportPixelX = -g_mapTilePixelWidth;
        m_viewportPixelY = -g_mapTilePixelHeight;
    }
    else
    {
        m_viewportPixelX = 14 - g_mapTilePixelWidth;
        m_viewportPixelY = 12 - g_mapTilePixelHeight;
    }

    Rect bounds;
    SetRect(&bounds,
            m_viewportPixelX,
            m_viewportPixelY,
            m_viewportPixelX + (m_visibleTileColumns + 1) * g_mapTilePixelWidth,
            m_viewportPixelY + (m_visibleTileRows + 1) * g_mapTilePixelHeight);
    SetBounds(&bounds);

    m_tileOriginX = targetX - oldViewport.column;
    m_tileOriginY = targetY - oldViewport.row;

    if (m_objectList != NULL)
        m_objectList->UpdateForViewportOrigin(m_tileOriginX,
                                              m_tileOriginY);

    DrawVisibleTiles();

    MapRect visible;
    GetClampedVisibleTileBounds(&visible);

    unsigned int checksum = ComputeVisibleTileChecksum(visible);

    unsigned char packet[11];
    unsigned char *out = packet;
    *out++ = 0x05;
    PacketBufferWriteUInt16BE(static_cast<unsigned short>(visible.left), out);
    out += 2;
    PacketBufferWriteUInt16BE(static_cast<unsigned short>(visible.top), out);
    out += 2;
    PacketBufferWriteUInt8(
        static_cast<unsigned char>(oldViewport.column - visible.left), out);
    out += 1;
    PacketBufferWriteUInt8(
        static_cast<unsigned char>(oldViewport.row - visible.top), out);
    out += 1;
    PacketBufferWriteUInt24BE(checksum, out);
    out += 3;
    packet[10] = 0;

    g_packetSender->QueueAndSendPacket(packet, 10);

    MapRect objectRefreshBounds;
    GetExtendedVisibleTileBounds(&objectRefreshBounds);
    RefreshStaticObjectTileRegion(objectRefreshBounds);
}

## Behavior

The method accepts target map coordinates plus a viewport-offset pair. When the final flag is set, it ignores the supplied offset pair and reads the active local-player viewport row/column through [UID:0002RW][0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition](by-memory/0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition.md). [UID:0002RW][0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition](by-memory/0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition.md) resolves the getter's raw output order as row/y in element 0 and column/x in element 1. The local-object create/update caller is the only modeled caller that passes this flag set. The packet-dispatch, UserPane, movement-correction, and unmodeled code-reference callers pass the flag clear and supply the offset pair directly.

After selecting the offset pair, the method validates both target coordinates against the stored map dimensions and validates the offset pair against the visible viewport extents. Invalid coordinates return without side effects.

The valid path recalculates the MapPane pixel origin from [UID:0000SW][g_useEpfAssets](by-global/g_useEpfAssets.md) and [UID:0000T7][MapTilePixelDimensions](by-global/MapTilePixelDimensions.md). Current-layout mode stores negative tile-width/tile-height offsets; the alternate layout stores `14 - tileWidth` and `12 - tileHeight`. It then builds a rectangle covering `(viewportWidth + 1) * tileWidth` by `(viewportHeight + 1) * tileHeight`, stores it through [UID:00015S][0x004b7c30-0x004b7f87.RectGeometryHelpers](by-memory/0x004b7c30-0x004b7f87.RectGeometryHelpers.md) and [UID:0001FW][0x00554680-0x00554b38.RegionAndMotionRect](by-memory/0x00554680-0x00554b38.RegionAndMotionRect.md), and invokes the MapPane bounds/update virtual slot with the pane-local bounds object.

The tile origin fields are then set to `targetX - offsetX` and `targetY - offsetY`. B010's ObjectList source-quality pass corrects the ObjectList call at `0x00505b20`: this method passes the new absolute tile origin values to [UID:00023F][0x00532f70-0x0053728e.ObjectListExtendedTypeLookupHelpers](by-memory/0x00532f70-0x0053728e.ObjectListExtendedTypeLookupHelpers.md)'s `ObjectList::UpdateForViewportOrigin(int originX, int originY)`, not precomputed origin deltas. The later MapPane tile/object-grid path at `0x0050bf61` is the caller that passes origin deltas to `ObjectList::ShiftObjectBucketsForOriginDelta`. After the absolute-origin ObjectList update, this method redraws the visible tile cache through the MapPane draw helper in [UID:0001AT][0x00506970-0x0050e320.MapPaneInputPacketRenderCore](by-memory/0x00506970-0x0050e320.MapPaneInputPacketRenderCore.md), computes a clamped visible tile rectangle, and calculates a 16-bit checksum over that range using [UID:00027N][0x0066db4c-0x0066dd50.MapPaneChecksumLookupTable](by-memory/0x0066db4c-0x0066dd50.MapPaneChecksumLookupTable.md).

The method sends an opcode `0x05` position packet through [UID:0001P0][0x0067a7ec-0x0067a7f0.g_packetSender](by-memory/0x0067a7ec-0x0067a7f0.g_packetSender.md) and [UID:0001HU][0x00574bb0-0x00574c13.QueueAndSendPacket](by-memory/0x00574bb0-0x00574c13.QueueAndSendPacket.md). After sending, it computes a wider object refresh rectangle and calls the nearby MapPane/ObjectList reconciliation helper at `0x0050f130`, which removes or creates object panes when visible-tile object ids no longer match the current [UID:00009Q][ObjectList](by-class/ObjectList.md) contents.

The previous flag-clear packet-delta caveat is now best treated as caller/ABI/decompiler noise rather than an unresolved semantic blocker. The defensible source shape is that the supplied viewport-offset pair initializes the old/current viewport pair for flag-clear callers; when the final flag is set, the method replaces that pair with the active object's cached viewport position through [UID:0002RW][0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition](by-memory/0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition.md). All known flag-clear callers pass the offset pair and execute the send path, so leaving the delta fields as uninitialized source locals is not credible. [UID:0002RW][0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition](by-memory/0x005a2de0-0x005a2dfb.UserPaneGetViewportPosition.md) closes the active-object getter order question: element 0 is row/y and element 1 is column/x. This page's existing `MapPoint` and `viewportX`/`viewportY` source-facing names stay as accepted first-draft notation until a broader shared point-type declaration pass is done.

## Reconstruction Notes

Formal first-draft C++ is recommended under the active code-entry gate and calls `ObjectList::UpdateForViewportOrigin(m_tileOriginX, m_tileOriginY)` with absolute origin values. The source-facing signature is `MapPane::RecenterAndSendPosition(int targetX, int targetY, int viewportX, int viewportY, bool useActiveObjectViewport)`. Canonical fields are `m_mapWidth/m_mapHeight` at `+0x3f4/+0x3f6`, `m_visibleTileColumns/m_visibleTileRows` at `+0x404/+0x408`, inherited `m_viewportPixelX` at `+0xb0` and `m_viewportPixelY` at `+0xac`, `m_tileOriginX/m_tileOriginY` at `+0x3fc/+0x400`, `m_localPlayerObjectPane` at `+0x418`, and `m_objectList` at `+0x424`. Outbound opcode `0x05` belongs to this MapPane send helper, not to the inbound `MapServerPacketOpcode` dispatcher enum. Packet layout remains byte `0` opcode, bytes `1..4` clamped visible-range coordinates, bytes `5..6` viewport deltas from the visible origin, bytes `7..9` checksum, and a zeroed staging byte outside the 10-byte send length. Exact names for the ObjectList recenter helper and checksum helper remain descriptive cleanup items, not C++ blockers.
