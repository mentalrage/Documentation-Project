typedef mystr::StringBase<char, mystr::mychar_traits<char> > SoundPathString;

void SoundManager::ScanMusicDirectory(
    const mystr::StringBase<wchar_t, mystr::mychar_traits<wchar_t> > &directoryPath,
    bool randomize,
    bool recursive)
{
    m_localMusicPaths.clear();

    if (directoryPath.length() == 0)
        return;

    std::vector<SoundPathString> musicPaths;
    SoundPathString rootDirectory(directoryPath);
    std::deque<SoundPathString> directories;
    directories.push_back(rootDirectory);

    while (!directories.empty()) {
        SoundPathString currentDirectory;
        currentDirectory = directories.front();
        directories.pop_front();

        WIN32_FIND_DATAA findData;
        SoundPathString searchPath = currentDirectory + "\\*";
        HANDLE findHandle = FindFirstFileA(searchPath.c_str(), &findData);

        if (findHandle != INVALID_HANDLE_VALUE) {
            do {
                SoundPathString fileName(findData.cFileName);

                if (fileName.ToLower().EndsWithAnsi(".mp3") &&
                    (findData.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) == 0) {
                    musicPaths.push_back(currentDirectory + "/" + fileName);
                }

                if (recursive &&
                    fileName.c_str()[0] != '.' &&
                    (findData.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) != 0) {
                    directories.push_back(currentDirectory + "/" + fileName);
                }
            } while (FindNextFileA(findHandle, &findData));
        }
    }

    m_localMusicPaths.reserve(musicPaths.size());

    if (randomize) {
        while (!musicPaths.empty()) {
            const std::vector<SoundPathString>::size_type index =
                rand() % musicPaths.size();
            m_localMusicPaths.push_back(musicPaths[index]);
            musicPaths.erase(musicPaths.begin() + index);
        }
    } else {
        m_localMusicPaths.swap(musicPaths);
    }
}

## Detailed Behavior

1. Clear `m_localMusicPaths` before any path test. If the wide input has zero length, return with the member vector empty.
2. Convert the root path once to ANSI `SoundPathString`. Create an automatic `std::deque<SoundPathString>` and push the root directory.
3. While the deque is not empty, copy its front directory and pop that entry.
4. Build `currentDirectory + "\\*"`, then call `FindFirstFileA` into an automatic `WIN32_FIND_DATAA`.
5. For each enumerated entry, construct an ANSI `SoundPathString` from `cFileName` and lowercase a copy through UID0004FM's StringBase member instantiation.
6. If the lowercase copy ends with `.mp3` and the directory attribute bit is clear, append `currentDirectory + "/" + fileName` to the temporary result vector.
7. Independently, when recursion is enabled, the first filename character is not `.`, and the directory attribute bit is set, append the same slash-joined path to the pending-directory deque.
8. Continue with `FindNextFileA` until it returns false. The original target never calls `FindClose`; reconstruction intentionally preserves that omission.
9. Reserve `m_localMusicPaths` to the complete temporary result count before either finalization branch.
10. In random mode, repeatedly compute `rand() % musicPaths.size()`, append that element to the member vector, and erase it from the temporary vector until empty. The binary uses global unseeded `rand`; no clamp, seed, or alternate RNG belongs here.
11. In nonrandom mode, swap the temporary vector's pointer triplet with `m_localMusicPaths`.
12. Let automatic vector, deque, and StringBase destructors regenerate the cleanup and exception paths.

Dot-prefixed names are excluded only from recursive directory queueing. The `.mp3` file test remains independent. Slash `/` is the observed join separator for both accepted files and queued subdirectories.
