void Application::RunMessageLoop()
{
    MSG message;

    g_pEventDispatcher->CaptureIdleTickBaseline();

    for (;;) {
        if (++g_applicationMessageLoopThrottleCounter > 10) {
            Sleep(1);
            g_applicationMessageLoopThrottleCounter = 0;
        }

        if (++g_applicationHeartbeatCheckCounter > 1000) {
            CheckTimerSkewAndSendHeartbeat();
            g_applicationHeartbeatCheckCounter = 0;
        }

        DWORD waitResult = MsgWaitForMultipleObjects(1, &m_hShutdownEvent, FALSE, 5, QS_ALLINPUT);
        if (waitResult == WAIT_OBJECT_0) {
            return;
        }

        if (waitResult == WAIT_TIMEOUT) {
            g_pEventDispatcher->ProcessIdleWork();
            continue;
        }

        if (waitResult != WAIT_OBJECT_0 + 1) {
            return;
        }

        while (g_pfnPeekMessage(&message, NULL, 0, 0, PM_REMOVE)) {
            if (message.message == WM_QUIT) {
                return;
            }

            if (g_activeBrowserControlPane != NULL &&
                BrowserControlPaneTranslateAcceleratorMessage(&message) == 0) {
                continue;
            }

            int inputMode = g_pInputMan->GetInputMode();
            if (inputMode == 2 || inputMode == 1) {
                TranslateMessage(&message);
            }

            MSGHandler *handler;
            {
                CriticalSectionLock lock(&s_messageHandlerCriticalSection);
                handler = m_activeMessageHandler;
            }

            handler->DispatchMessage(&message);
        }

        g_pEventDispatcher->ProcessIdleWork();
    }
}

## Behavior

Runs the main Win32 message loop around `MsgWaitForMultipleObjects`, drains pending messages through the PlatformApi `PeekMessageW` slot, checks browser/input gates before dispatch, snapshots the active message handler under the Application critical section, runs idle work on timeout or drained queue, and periodically calls the timer-skew/heartbeat helper.

The live loop shape is:

- snapshots the EventDispatcher idle baseline once through [UID:000143][0x004a7120-0x004a712c.ApplicationIdleTickBaseline](by-memory/0x004a7120-0x004a712c.ApplicationIdleTickBaseline.md);
- every iteration increments the throttle counter in [UID:00029B][0x0067ab54-0x0067ab84.ApplicationDialogTimerAndBackgroundGlobals](by-memory/0x0067ab54-0x0067ab84.ApplicationDialogTimerAndBackgroundGlobals.md); when it exceeds `10`, the loop sleeps for one millisecond and resets the counter;
- increments the heartbeat/timer counter from the same global-data cluster; when it exceeds `1000`, the loop calls [UID:0000YT][0x00466ca0-0x004670ad.CheckTimerSkewAndSendHeartbeat](by-memory/0x00466ca0-0x004670ad.CheckTimerSkewAndSendHeartbeat.md) at `0x00464d26` and resets the counter;
- waits on one handle, the Application shutdown event at object offset `0x83c`, with a five-millisecond timeout and queue mask `0x4ff`;
- returns immediately when the shutdown event is signaled, when `WM_QUIT` is drained, or when `MsgWaitForMultipleObjects` returns an unexpected value;
- treats `WAIT_OBJECT_0 + 1` as the message-queue branch and `WAIT_TIMEOUT` / `0x102` as the idle-only branch;
- on queue activity, drains messages with the PlatformApi `PeekMessageW` dispatch entry at `0x0069be8c` using `PM_REMOVE`;
- returns on `WM_QUIT` before browser/input/handler dispatch;
- gives the active browser-control pane a chance to consume keyboard/accelerator messages before translation and dispatch; a zero return from `BrowserControlPaneTranslateAcceleratorMessage(&message)` means the message was consumed and the loop continues to the next queued message;
- calls `InputMan` input-mode state and only calls `TranslateMessage` for modes `1` and `2`;
- uses `CriticalSectionLock` with `s_messageHandlerCriticalSection`, snapshots `m_activeMessageHandler`, releases the lock, and then calls `handler->DispatchMessage(&message)`;
- deliberately performs no active-handler null check because the binary calls the virtual slot unconditionally after the snapshot;
- calls [UID:000142][0x004a6c40-0x004a6cda.ApplicationIdleWorkScheduler](by-memory/0x004a6c40-0x004a6cda.ApplicationIdleWorkScheduler.md) when the queue drains or when the wait times out.
