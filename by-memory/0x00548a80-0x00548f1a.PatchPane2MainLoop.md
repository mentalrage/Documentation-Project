bool PatchPane2::HandleDownloadWorkEvent(const WorkThreadNotification *event)
{
    if (event->m_channel != kHttpNotification)
        return false;

    const httpget::Notification *notification =
        static_cast<const httpget::Notification *>(event->m_payload);

    switch (notification->m_state) {
    case httpget::Started:
    {
        m_activePatchFile = _wfopen(m_tempPatchPath.c_str(), L"wb");
        if (m_activePatchFile == 0)
            ReportWriteError(m_tempPatchPath.c_str());

        GetChild<StaticTextControlPane>(0)->SetText(m_finalPatchPath.c_str());

        StringBase<wchar_t> byteCount;
        byteCount.FormatWide(L"0", ftell(m_activePatchFile));
        GetChild<StaticTextControlPane>(1)->SetText(byteCount.c_str());
        return true;
    }

    case httpget::DataChunk:
    {
        const _AUTOBUF<unsigned char> *chunk =
            static_cast<const _AUTOBUF<unsigned char> *>(notification->m_payload);

        if (fwrite(chunk->data(), 1, chunk->size(), m_activePatchFile) != chunk->size())
            ReportWriteError(m_tempPatchPath.c_str());

        StringBase<wchar_t> byteCount;
        byteCount.FormatWide(L"%d", ftell(m_activePatchFile));
        GetChild<StaticTextControlPane>(1)->SetText(byteCount.c_str());
        return true;
    }

    case httpget::Completed:
    {
        FILE *file = m_activePatchFile;
        const int closeResult = fclose(file);
        m_activePatchFile = 0;
        if (closeResult == EOF)
            throw Win32Error();

        StringBase<char> finalPath(m_finalPatchPath);
        _unlink(finalPath.c_str());

        StringBase<char> renameTarget(m_finalPatchPath);
        StringBase<char> renameSource(m_tempPatchPath);
        if (rename(renameSource.c_str(), renameTarget.c_str()) != 0 && errno != ENOENT)
            throw Win32Error();

        SHELLEXECUTEINFOA executeInfo = {};
        executeInfo.cbSize = sizeof(executeInfo);
        executeInfo.lpVerb = "runas";
        executeInfo.lpFile = "patcher.exe";
        executeInfo.lpParameters = Locale;
        executeInfo.nShow = SW_SHOWNORMAL;
        ShellExecuteExA(&executeInfo);

        FinishAndExit(GetLanguageText(174));
        return true;
    }

    case httpget::ReadOrOpenUrlFailed:
        ClosePatchFile();

        if (m_currentDownloadIndex < m_patchFileNames.size() - 1) {
            ++m_currentDownloadIndex;
            SendDownloadRequest();
            return true;
        }

        if (m_downloadPassCount >= 10)
            FinishAndExit(GetLanguageText(157));
        else {
            m_currentDownloadIndex = 0;
            ++m_downloadPassCount;
        }

        SendDownloadRequest();
        return true;

    case httpget::InternetOpenFailed:
    {
        ClosePatchFile();

        wchar_t message[1024];
        const Error *error = static_cast<const Error *>(notification->m_payload);
        error->FormatErrorMessage(message, 1024);
        FinishAndExit(message);
        return true;
    }

    default:
        return true;
    }
}

## Behavior

This is the central `PatchPane2` work-thread callback. It validates the incoming work event magic, switches on the event subtype, and coordinates file open/write/close/rename, progress text, next-file requests, completion, and error exit.

| Event subtype | Work payload / state touched | Behavior |
| --- | --- |
| `0` | Opens owner `+0x270` temp path, stores `FILE*` at owner `+0x284`, updates controls `0` and `1`. | Opens the output/temp patch file, reports write/open failures through `0x005492d0`, shows the final path in the status control, and initializes byte-count text from `ftell`. |
| `1` | Uses payload object at event-data `+0x08`; data pointer is payload `+0x04`, length is payload `+0x08`. | Writes a received chunk to the active file with `_fwrite`, reports short writes through `0x005492d0`, and updates byte-count text from `ftell`. |
| `2` | Closes owner `+0x284`, deletes final path, renames temp path to final path, launches external patcher. | Finalizes the patcher download, throws `Win32Error` on close/rename failure, launches `patcher.exe` with verb `runas`, then exits through `FinishAndExit` with language id `174`. |
| `3` | Closes owner `+0x284`, updates current file index `+0x280` and retry/pass counter `+0x288`. | Advances to the next file request, wraps to index `0` for another pass, or after the observed `0x0a` pass threshold calls the finish/error path with language id `157` and then still falls through to send another request through `0x00549220`. |
| `4` | Reads a wide error string from the event payload into a 1024-wide-character stack buffer. | Closes any active file and exits through `FinishAndExit` with the worker-provided error text. |

The callback returns `0` only when the incoming work-event magic at `event + 0x08` is not `0x68747470` (`"http"` in little-endian form). Recognized subtype paths return `1`.
