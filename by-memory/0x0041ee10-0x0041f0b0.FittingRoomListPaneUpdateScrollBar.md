void FittingRoomListPane::UpdateScrollBar()
{
    const int itemCount = m_catalog->GetItemCount(m_category);
    m_visibleItemCount = CountVisibleItems(m_category);

    const int pageSize = m_displayMode == ListMode ? 26 : 3;
    const int logicalCount = m_displayMode == ListMode ? itemCount : (itemCount + 5) / 6;
    const short maxScroll = Clamp(logicalCount - pageSize, 0, 30000);

    m_scrollPane->SetRange(maxScroll);
    m_scrollPane->SetVisible(maxScroll > 0);
    m_scrollPosition = Clamp(m_scrollPosition, 0, maxScroll);
}
