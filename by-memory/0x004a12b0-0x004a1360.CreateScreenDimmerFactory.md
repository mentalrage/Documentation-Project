static Pane *GetUserPaneActiveDialogHostPane(UserPane *userPane)

    return *reinterpret_cast<Pane **>(
        reinterpret_cast<unsigned char *>(userPane) +
        kCollectionDataActiveDialogHostPaneOffset);

ScreenDimmer *CreateScreenDimmer(void)

    Pane *parentPane;

    }
