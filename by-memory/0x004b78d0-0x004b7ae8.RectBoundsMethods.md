{
    if (bottom <= top || right <= left ||
        bounds->bottom <= bounds->top || bounds->right <= bounds->left) {
        left = 0;
        top = 0;
        right = 0;
        bottom = 0;
        return false;

        top = 0;
        right = 0;
        bottom = 0;
        return false;

    top = newTop;
    right = newRight;
    bottom = newBottom;
    return true;
}
