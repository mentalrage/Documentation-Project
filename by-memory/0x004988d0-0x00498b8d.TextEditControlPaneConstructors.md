    const short contentWidth = static_cast<short>(width * textWidthScale);

                                      textBackColor,
                                      hasHorizontalScrollbar,

                                      styleFlags,
                                      IsLegacyAssetMode(),
                                      true,
                                      useThousandsSeparator);

    RectBounds childBounds = *bounds;
    OffsetRect(&childBounds, -childBounds.left, -childBounds.top);
