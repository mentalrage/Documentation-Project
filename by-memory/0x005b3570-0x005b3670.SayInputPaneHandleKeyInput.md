bool SayInputPane::HandleKeyOrTextEvent(Event *event)
{
    const unsigned char key =
        g_pEventMan->TranslateEventKey(
            event->m_payload.m_key.m_key,
            event->m_payload.m_key.m_modifiers);

    if (event->m_type != kEventKeyDown ||
        event->m_payload.m_key.m_modifiers != 0 ||
        (key != kPaneKeyUp && key != kPaneKeyDown)) {
        return LineInputPane::HandleKeyOrTextEvent(event);
    }

    wchar_t historyText[256];
    if (key == kPaneKeyUp) {
        g_pUserPane->GetPreviousChatHistoryLine(
            &m_chatHistoryCursor, historyText, _countof(historyText));
    } else {
        g_pUserPane->GetNextChatHistoryLine(
            &m_chatHistoryCursor, historyText, _countof(historyText));
    }

    if (historyText[0] != L'\0') {
        m_textEditPane->SetSelectionRange(0, 0x7fff);
        m_textEditPane->InsertTextRange(
            historyText,
            static_cast<short>(wcslen(historyText)),
            NULL,
            NULL);
        m_textEditPane->SetSelectionRange(0, 0x7fff);
    }

    return true;
}

## Behavior

The handler intercepts previous/next chat-history keys and otherwise delegates to the base line-input handler:

1. Calls `g_pEventMan->TranslateEventKey(event key, modifiers)` before any Event/modifier/key gate.
2. Intercepts only `kEventKeyDown`, modifiers exactly zero, and translated `kPaneKeyUp` or `kPaneKeyDown`.
3. Allocates an uninitialized local `wchar_t historyText[256]`; each history helper always writes either an empty terminator or a bounded copied line, so source does not need a synthetic zero-fill.
4. Calls `g_pUserPane->GetPreviousChatHistoryLine` for Up or `GetNextChatHistoryLine` for Down, passing the signed-short cursor by address and `_countof(historyText)`.
5. If output is nonempty, calls inherited `m_textEditPane->SetSelectionRange(0, 0x7fff)`, inserts the line through `InsertTextRange` with a signed-short length and null format/object pointers, then selects the same range again.
6. Returns handled (`true`) for Up/Down even when history output is empty.
7. Calls `LineInputPane::HandleKeyOrTextEvent(event)` for every other Event.
