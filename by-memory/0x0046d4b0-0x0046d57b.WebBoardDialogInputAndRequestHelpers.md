char __thiscall WebBoardDialog::OnInputKey(KeyState *key)
{
    char ch = std::ctype<char>::do_narrow(key->rawCharOrVirtualKey, key->modifierByte);

    if (key->eventType == 8) {
        if (ch == 'b' && (key->modifierByte & 7) == 0) {
