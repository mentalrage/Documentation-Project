        Rgb565AlphaMapMmxBlocks(
            pixels,
            pixelPitchBytes,
            alphaBytes,
            alphaPitchBytes,
            blockCount4Pixels,
            rowCount);

        pixels += blockCount4Pixels * 4;
        alphaBytes += blockCount4Pixels * 4;

            const unsigned int pixel = pixels[pixelIndex];

                (redBlue & 0xf81f) | (green & 0x07e0));
