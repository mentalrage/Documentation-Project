            const ExchangeItemListRow *existing =
                static_cast<const ExchangeItemListRow *>(list->GetRow(i));

                list->RemoveRows(i, 1);
                insertIndex = i;
                break;
            }
        }

        ExchangeItemListRow row;
        memset(&row, 0, sizeof(row));
        row.key = rowKey;
        row.itemId = itemId;
        row.color = color;
        wcscpy_s(row.name, 256, nameText);
        list->InsertRow(insertIndex, &row);
        return true;

## Behavior
