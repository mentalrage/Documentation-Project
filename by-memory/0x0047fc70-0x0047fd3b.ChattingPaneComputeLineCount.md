## Behavior

It reads the message list pointer at `this + 0x130`, iterates `list->count` at `list + 0x0c`, fetches each message through list vtable slot `+0x10`, and sums each message's measured row count through the message virtual slot `+0x10`.

int ChattingPane::ComputeLineCount() const
{
    const int width = (g_useEpfAssets == 1) ? 783 : 408;
    int total = 0;

    for (int i = 0; i < m_messages->GetCount(); ++i) {
        ColorStringChattingMessage *message = m_messages->GetAt(i);
        if (message != nullptr)
            total += message->MeasureRows(this, width);
    }

    return total;
}
