void NumericStringControlPane::OnPaint()
{
    RectBounds bounds;
    GetBounds(&bounds);
    SetDrawColor(0);
    FillRect(&bounds);

    const short textWidth =
        static_cast<short>(9 * wcslen(m_text.c_str()));

    int glyphLeft;
    if (m_alignment == 1) {
        glyphLeft = bounds.right - textWidth;
    } else if (m_alignment == 2) {
        glyphLeft =
            bounds.left + (bounds.right - textWidth - bounds.left) / 2;
    } else {
        glyphLeft = bounds.left;
    }

    RectBounds glyphBounds;
    InitRectBounds(&glyphBounds,
                   glyphLeft,
                   bounds.top,
                   glyphLeft + 9,
                   bounds.top + 11);

    EPFTileContext glyphContext;
    for (const wchar_t *character = m_text.c_str();
         *character != L'\0';
         ++character) {
        if (*character == L'-') {
            g_pEPFLib->LookupLayoutEntry(L"FONTSYMB.EPF",
                                         4,
                                         &glyphContext);

            glyphBounds.Offset(glyphContext.bounds.left,
                               glyphContext.bounds.top);
            RenderTileFrame(&glyphContext,
                            &glyphContext.bounds,
                            &glyphBounds,
                            1,
                            L"FONTSYMB.PAL",
                            NULL);
            glyphBounds.Offset(-glyphContext.bounds.left,
                               -glyphContext.bounds.top);
        } else if (*character != L' ' && *character != L'/') {
            InitNumericGlyphTileContext(&glyphContext,
                                        m_numericGlyphs,
                                        *character,
                                        m_colorVariant);
            g_pfnBlitSprite(this,
                            &glyphContext,
                            &glyphContext.bounds,
                            &glyphBounds,
                            1,
                            NULL,
                            NULL);
        }

        OffsetRect(&glyphBounds, 9, 0);
    }
}

## Behavior

- Copies inherited bounds through `GetBounds`, calls `SetDrawColor(0)`, and uses the accepted control-source `FillRect(&bounds)` wrapper for Surface callback slot `0x0069b3fc`.
- Calls `m_text.c_str()`, measures UTF-16 length, multiplies by `9`, and truncates to signed `short textWidth`; a second `c_str()` call supplies the iteration pointer, matching the two live StringBase accessor calls.
- Uses `m_alignment` at `+0x10c`: mode `1` computes `bounds.right - textWidth`, mode `2` computes `bounds.left + (bounds.right - textWidth - bounds.left) / 2`, and all other values start at `bounds.left`.
- Initializes a destination `RectBounds` at `(glyphLeft, bounds.top, glyphLeft + 9, bounds.top + 11)` and iterates one UTF-16 code unit at a time.
- For `'-'`, initializes a stack `EPFTileContext`, looks up frame `4` from `FONTSYMB.EPF`, offsets the destination by the returned source bounds, calls `RenderTileFrame` with mode `1`, `FONTSYMB.PAL`, and null options, then reverses the offset.
- For space and slash, performs no draw. Both still reach the unconditional `OffsetRect(&glyphBounds, 9, 0)`.
- For all other code units, calls [UID:0003XD][0x004bb1e0-0x004bb252.NumericStringGlyphTileContextInit](by-memory/0x004bb1e0-0x004bb252.NumericStringGlyphTileContextInit.md) with `m_numericGlyphs`, the code unit, and sign-extended `m_colorVariant`, then dispatches `g_pfnBlitSprite` with mode `1`, null palette, and null options.
- There is no null check for `m_numericGlyphs`, no `LookupLayoutEntry` result check, no fallback glyph/resource branch, and no explicit tile-context release in this method. These absences are preserved rather than normalized.
