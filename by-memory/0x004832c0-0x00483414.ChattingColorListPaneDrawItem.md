```cpp
void ChattingColorListPane::DrawItem(int row, const ColorItem& item, const Rect& rect)
{
    if (item.colorValue == 0)
        Fill(rect.Inset(2), PaletteIndex(136));

    if (IsSelected(row)) {
        Fill(rect, PaletteIndex(128));
        Fill(rect.Inset(2), PaletteIndex(143));
    }

    Fill(rect.Inset(3), item.colorValue);
}
```
