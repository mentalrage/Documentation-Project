void SayInputPane::OnSubmitInput()
{
    const short textLength = LineInputPane::TextLength();
    if (textLength <= 0) {
        return;
    }

    if (g_pCollectionData->chatInputSendBlocked != 0) {
        return;
    }

    wchar_t wideText[256] = {};
    LineInputPane::CopyText(wideText, 255);
    if (wideText[0] == L'\0') {
        return;
    }

    ValidateChatInputOrThrowPasswordError(wideText);
    SanitizeWideTextForChat(wideText, static_cast<int>(wcslen(wideText)));

    const unsigned char kChatMessageOpcode = 0x0e;
    const unsigned char kSayChatSubtype = 0;

    unsigned char packet[300] = {};
    char multibyteText[256] = {};
    void *packetCopyContext = GetMemoryMan();

    PacketBufferWriteUInt8(kChatMessageOpcode, packet + 0);
    PacketBufferWriteUInt8(kSayChatSubtype, packet + 1);

    const int convertedLength = WideCharToMultiByte(
        0, 0, wideText, static_cast<int>(wcslen(wideText)),
        multibyteText, sizeof(multibyteText), 0, 0);
    if (convertedLength >= 0x100) {
        ReportRangeCheckFailure();
    }

    multibyteText[convertedLength] = '\0';
    PacketBufferWriteUInt8(static_cast<unsigned char>(convertedLength), packet + 2);
    CopyPacketBytes(packetCopyContext, packet + 3, multibyteText, convertedLength);

    const int packetLength = convertedLength + 3;
    if (packetLength >= 0x12c) {
        ReportRangeCheckFailure();
    }

    packet[packetLength] = 0;
    QueueAndSendPacket(g_packetSender, packet, packetLength);
    g_pUserPane->SaveChatHistoryLine(wideText);
}

## Behavior

The handler sends the normal local say-chat message:

1. Reads current input length and returns without side effects when it is not positive.
2. Checks the local-player/chat-input send gate at `g_pCollectionData+0x134102`; nonzero blocks sending.
3. Copies up to `255` wide characters from the inherited LineInputPane text control.
4. Rejects empty copied text.
5. Runs the password-disclosure guard and chat text sanitizer.
6. Builds a packet with opcode `0x0e`, subtype `0`, one-byte converted payload length, and converted multibyte text.
7. Range-checks converted length `< 0x100` and final packet length `< 0x12c`.
8. Sends exactly `convertedLength + 3` bytes through `QueueAndSendPacket(g_packetSender, ...)`.
9. Saves the submitted wide string through `g_pUserPane->SaveChatHistoryLine(wideText)`; the receiver-bearing helper and `m_chatHistory` list belong to UserPane, while this method remains the SayInputPane submit owner.

The scratch zero bytes written after packet construction and payload conversion are local safety/terminator bytes outside the explicit send length; they are not additional protocol fields.
