void ScreenPane::PrepareDirectDrawPresentation()
{
    if (g_pDirectX == NULL)
        return;

    LPDIRECTDRAW directDraw = g_pDirectX->m_pDirectDraw;

    if (!m_restoreCooperativeLevelPending) {
        DWORD cooperativeFlags = ApplicationActiveGate
            ? DDSCL_NORMAL
            : (DDSCL_FULLSCREEN | DDSCL_ALLOWREBOOT | DDSCL_EXCLUSIVE);

        HRESULT hr = directDraw->SetCooperativeLevel(
            g_pApplication->GetMainWindowHandle(),
            cooperativeFlags);
        if (FAILED(hr))
            throw new DDError(hr);

        m_restoreCooperativeLevelPending = true;
    }

    if (!m_restoreDisplayModePending) {
        if (ApplicationActiveGate == 0) {
            RECT desktopRect;
            GetWindowRect(GetDesktopWindow(), &desktopRect);

            HRESULT hr = directDraw->SetDisplayMode(
                m_screenWidth,
                m_screenHeight,
                m_screenBitDepth);
            if (FAILED(hr))
                throw new DDError(hr);
        }

        m_restoreDisplayModePending = true;
    }

    if (m_primarySurface != NULL) {
        RECT fillRect;
        fillRect.left = 0;
        fillRect.top = 0;
        fillRect.right = m_screenWidth;
        fillRect.bottom = m_screenHeight;

        DDBLTFX bltFx;
        memset(&bltFx, 0, sizeof(bltFx));
        bltFx.dwSize = sizeof(bltFx);
        bltFx.dwFillColor = 0;

        if (ApplicationActiveGate) {
            OffsetRect(
                &fillRect,
                g_pApplication->m_windowOffsetX,
                g_pApplication->m_windowOffsetY);
        }

        HRESULT hr = m_primarySurface->Blt(
            &fillRect,
            NULL,
            NULL,
            DDBLT_COLORFILL,
            &bltFx);

        if (hr == DDERR_SURFACELOST) {
            m_primarySurface->Restore();
            m_primarySurface->Blt(
                &fillRect,
                NULL,
                NULL,
                DDBLT_COLORFILL,
                &bltFx);
        }
    }
}

## Behavior

The method returns immediately when [UID:0000QS][g_pDirectX](by-global/g_pDirectX.md) is null. Otherwise it loads `g_pDirectX->m_pDirectDraw` and prepares the root screen object's DirectDraw presentation state:

- If byte `+0x110` is clear, it calls `IDirectDraw::SetCooperativeLevel(g_pApplication->GetMainWindowHandle(), flags)`, with `DDSCL_NORMAL` (`0x08`, 8 decimal, Verified with MCP `int_convert`) when `ApplicationActiveGate` is set and `DDSCL_FULLSCREEN | DDSCL_ALLOWREBOOT | DDSCL_EXCLUSIVE` (`0x13`, 19 decimal, Verified with MCP `int_convert`) when it is clear. On success it writes byte `+0x110 = 1` at `0x005574e5`.
- If byte `+0x111` is clear, it calls `GetDesktopWindow()` and `GetWindowRect()` when `ApplicationActiveGate == 0`, then calls `IDirectDraw::SetDisplayMode(m_screenWidth, m_screenHeight, m_screenBitDepth)` through vtable slot `+0x54`. On success, or immediately when the active gate skips the display-mode call, it writes byte `+0x111 = 1` at `0x00557537`.
- If `m_primarySurface` / `+0x100` is non-null, it builds a rectangle from ScreenPane dimensions `+0x112/+0x114`, zeroes a `DDBLTFX`-sized block, sets `dwSize = 100`, sets fill color to zero, applies application window offsets `g_pApplication + 0xa5c` and `+0xa58` while active, and calls `IDirectDrawSurface::Blt(&rect, NULL, NULL, DDBLT_COLORFILL, &bltFx)` through slot `+0x14`.
- If the first `Blt` returns the observed surface-lost HRESULT branch, it calls `IDirectDrawSurface::Restore()` through slot `+0x6c` and retries the same `Blt` once. The binary does not check the retry result.

The body throws pointer-to-[UID:00003L][DDError](by-class/DDError.md) objects on failed `SetCooperativeLevel()` and `SetDisplayMode()` paths through the project allocator, [UID:0002HK][0x004a6260-0x004a6284.DDErrorConstructor](by-memory/0x004a6260-0x004a6284.DDErrorConstructor.md), and `__CxxThrowException`. EH scope stores, security cookie checks, throw helper plumbing, and padding are compiler/runtime support, not source logic.
