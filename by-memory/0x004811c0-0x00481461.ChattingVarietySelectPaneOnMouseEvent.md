- `0`: hover/update path. It calls the six-row hit-test helper, stores the hovered row in `this + 0xf9`, returns whether a row is hot, and requests a repaint through the owning pane.

```cpp
bool ChattingVarietySelectPane::OnMouseEvent(MouseEvent const& event)
{
    if (g_client->modalBlockByte)
        return false;

    switch (event.kind) {
    case MouseEvent::Hover:
        m_hotRow = HitTest(event.x, event.y);
        RequestOwnerRepaint();
        return m_hotRow != 0xff;

    case MouseEvent::Press:
        if (PointInRect(event.x, event.y, m_bounds)) {
            m_pressed = true;
            RequestOwnerRepaint();
            return true;
        }
        RefreshChatVarietyButton();
        Dismiss();
        return false;

    case MouseEvent::Release:
        if (!m_pressed)
            return false;
        m_pressed = false;
        return CommitRowOrDismiss(HitTest(event.x, event.y), event.x, event.y);
    }

    return false;
}
```
