bool FittingRoomDialog::HandleFittingAction(int action, int, int)
{
    if (action == kPollCatalogRequest) {
        bool complete = false;

        if (m_catalogRequest != NULL) {
            FittingRoomDownloadControlPane* pane = downloadPane();
            const unsigned char percent =
                GetUsagePercent(m_catalogRequest->progress);

            if (pane->SetProgressPercent(percent))
                complete = percent >= 100;
        }

        if (m_catalogRequest == NULL || !m_catalogRequest->IsActive() || complete)
            StartTimer(kFittingRoomActionTimer, 100);

        return true;
    }

    if (action != kSubmitOrShowPreview)
        return true;

    if (IsCachedNameInvalid()) {
        StartTimer(kFittingRoomActionTimer, 100);
        return true;
    }

    if (NeedsCatalogRequestForCurrentAppearance()) {
        downloadPane()->SetWaiting(true);
        g_pCashShopRequest->SubmitCashShopCatalogRequest();
        StartTimer(kFittingRoomActionTimer, 100);
        return true;
    }

    ShowPreview();
    return true;
}
